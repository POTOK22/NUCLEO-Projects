/**
  * @brief  The application entry point.
  * @retval int
  */
int main(void)
{

  /* USER CODE BEGIN 1 */
	uint8_t pid = 0, ver = 0, midh = 0, midl = 0;
	uint32_t t0;
  /* USER CODE END 1 */

  /* MPU Configuration--------------------------------------------------------*/
  MPU_Config();

  /* MCU Configuration--------------------------------------------------------*/

  /* Reset of all peripherals, Initializes the Flash interface and the Systick. */
  HAL_Init();

  /* USER CODE BEGIN Init */

  /* USER CODE END Init */

  /* Configure the system clock */
  SystemClock_Config();

  /* USER CODE BEGIN SysInit */

  /* USER CODE END SysInit */

  /* Initialize all configured peripherals */
  MX_GPIO_Init();
  MX_DMA_Init();
  MX_ETH_Init();
  MX_I2C1_Init();
  MX_USART3_UART_Init();
  MX_DCMI_Init();
  /* USER CODE BEGIN 2 */
	LOG("\r\n\r\n=== OV7670 test - NUCLEO-F756ZG ===\r\n");
	LOG("HCLK      : %lu Hz\r\n", HAL_RCC_GetHCLKFreq());
	LOG("PCLK1     : %lu Hz\r\n", HAL_RCC_GetPCLK1Freq());
	LOG("Format    : %ux%u RGB565 (%u B/klatke)\r\n",
	OV7670_WIDTH, OV7670_HEIGHT, (unsigned) OV7670_FRAME_BYTES);

	LOG("Adres buf : %p\r\n", (void*) frame_buffer);

	/* XCLK (MCO na PA8) musi juz biegac zanim zwolnimy RESET# */
	HAL_Delay(50);
	OV7670_HardReset();

	/* --- diagnostyka magistrali ---------------------------------------- */
	I2C_ScanBus();

	if (OV7670_Probe(&pid, &ver, &midh, &midl) != HAL_OK) {
		LOG("[BLAD] Kamera nie odpowiada na SCCB (brak ACK).\r\n");
		LOG(
				"       Sprawdz: 3.3V, GND, pull-upy 4k7 na SDA/SCL, XCLK na PA8.\r\n");
		Error_Handler();
	}

	LOG("PID/VER   : 0x%02X / 0x%02X  (oczekiwane 0x76 / 0x73)\r\n", pid, ver);
	LOG("MID       : 0x%02X%02X       (oczekiwane 0x7FA2)\r\n", midh, midl);

	if (pid != 0x76) {
		LOG("[UWAGA] Nietypowy PID - to moze nie byc OV7670.\r\n");
	}

	/* --- konfiguracja sensora ------------------------------------------ */
	if (OV7670_Init() != HAL_OK) {
		LOG("[BLAD] Konfiguracja rejestrow nie powiodla sie.\r\n");
		Error_Handler();
	}
	LOG("Sensor    : skonfigurowany (QQVGA RGB565)\r\n");

#if TEST_COLOR_BAR
	OV7670_SetColorBar(1);
	LOG("Tryb      : PASKI TESTOWE\r\n");
#endif

	LOG("Start przechwytywania...\r\n\r\n");
  /* USER CODE END 2 */

  /* Infinite loop */
  /* USER CODE BEGIN WHILE */
	while (1) {
		frame_ready = 0;

		/* Dlugosc podajemy w SLOWACH 32-bitowych, nie w bajtach! */
		if (HAL_DCMI_Start_DMA(&hdcmi, DCMI_MODE_SNAPSHOT,
				(uint32_t) frame_buffer,
				OV7670_FRAME_WORDS) != HAL_OK) {
			LOG("[BLAD] HAL_DCMI_Start_DMA\r\n");
			HAL_Delay(500);
			continue;
		}

		t0 = HAL_GetTick();
		while (!frame_ready && (HAL_GetTick() - t0) < 1000) {
			/* czekamy na HAL_DCMI_FrameEventCallback */
		}

		HAL_DCMI_Stop(&hdcmi);

		if (!frame_ready) {
			LOG("[BLAD] Timeout - brak sygnalu VSYNC/HREF/PCLK.\r\n");
			LOG("       Sprawdz polaczenia DCMI i polaryzacje sygnalow.\r\n");
			LOG("DCMI SR    = 0x%08lX\r\n", hdcmi.Instance->SR);
			LOG("DCMI CR    = 0x%08lX\r\n", hdcmi.Instance->CR);
			LOG("DCMI ERR   = 0x%08lX\r\n", hdcmi.ErrorCode);

			LOG("DMA NDTR   = %lu\r\n", hdma_dcmi.Instance->NDTR);
			LOG("DMA CR     = 0x%08lX\r\n", hdma_dcmi.Instance->CR);
			LOG("DMA ERR    = 0x%08lX\r\n", hdma_dcmi.ErrorCode);
			LOG("DMA STATE  = 0x%08lX\r\n", hdma_dcmi.State);
			HAL_Delay(500);
			continue;
		}

		/* Jesli D-Cache jest wlaczony, dane zapisane przez DMA moga nie byc
		 * widoczne dla rdzenia - trzeba uniewaznic linie cache.           */
		if (SCB->CCR & SCB_CCR_DC_Msk) {
			SCB_InvalidateDCache_by_Addr((uint32_t*) frame_buffer,
					(int32_t) OV7670_FRAME_BYTES); /* <-- Zmiana tutaj! */
		}

		SendFrame();
		HAL_Delay(20);
    /* USER CODE END WHILE */

    /* USER CODE BEGIN 3 */
	}
  /* USER CODE END 3 */
}

/**
  * @brief System Clock Configuration
  * @retval None
  */
void SystemClock_Config(void)
{
  RCC_OscInitTypeDef RCC_OscInitStruct = {0};
  RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};

  /** Configure LSE Drive Capability
  */
  HAL_PWR_EnableBkUpAccess();

  /** Configure the main internal regulator output voltage
  */
  __HAL_RCC_PWR_CLK_ENABLE();
  __HAL_PWR_VOLTAGESCALING_CONFIG(PWR_REGULATOR_VOLTAGE_SCALE1);

  /** Initializes the RCC Oscillators according to the specified parameters
  * in the RCC_OscInitTypeDef structure.
  */
  RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_HSI|RCC_OSCILLATORTYPE_HSE;
  RCC_OscInitStruct.HSEState = RCC_HSE_BYPASS;
  RCC_OscInitStruct.HSIState = RCC_HSI_ON;
  RCC_OscInitStruct.HSICalibrationValue = RCC_HSICALIBRATION_DEFAULT;
  RCC_OscInitStruct.PLL.PLLState = RCC_PLL_ON;
  RCC_OscInitStruct.PLL.PLLSource = RCC_PLLSOURCE_HSE;
  RCC_OscInitStruct.PLL.PLLM = 4;
  RCC_OscInitStruct.PLL.PLLN = 216;
  RCC_OscInitStruct.PLL.PLLP = RCC_PLLP_DIV2;
  RCC_OscInitStruct.PLL.PLLQ = 3;
  if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK)
  {
    Error_Handler();
  }

  /** Activate the Over-Drive mode
  */
  if (HAL_PWREx_EnableOverDrive() != HAL_OK)
  {
    Error_Handler();
  }

  /** Initializes the CPU, AHB and APB buses clocks
  */
  RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK|RCC_CLOCKTYPE_SYSCLK
                              |RCC_CLOCKTYPE_PCLK1|RCC_CLOCKTYPE_PCLK2;
  RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_PLLCLK;
  RCC_ClkInitStruct.AHBCLKDivider = RCC_SYSCLK_DIV1;
  RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV4;
  RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV2;

  if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_7) != HAL_OK)
  {
    Error_Handler();
  }
  HAL_RCC_MCOConfig(RCC_MCO1, RCC_MCO1SOURCE_HSI, RCC_MCODIV_2);
}

/**
 * WAZNE: SCCB to nie jest pelne I2C.
 * Odczyt rejestru wymaga DWOCH osobnych transakcji zakonczonych STOP-em:
 *   1) zapis adresu rejestru  (STOP)
 *   2) odczyt jednego bajtu   (STOP)
 * Uzycie HAL_I2C_Mem_Read() (repeated START) NIE zadziala z ta kamera.
 */
static HAL_StatusTypeDef OV7670_WriteReg(uint8_t reg, uint8_t val) {
	uint8_t buf[2] = { reg, val };
	HAL_StatusTypeDef st;

	st = HAL_I2C_Master_Transmit(&hi2c1, OV7670_I2C_ADDR, buf, 2, 100);
	HAL_Delay(1); /* OV7670 lubi chwile przerwy miedzy zapisami */
	return st;
}

static HAL_StatusTypeDef OV7670_ReadReg(uint8_t reg, uint8_t *val) {
	HAL_StatusTypeDef st;

	st = HAL_I2C_Master_Transmit(&hi2c1, OV7670_I2C_ADDR, &reg, 1, 100);
	if (st != HAL_OK) {
		return st;
	}
	HAL_Delay(1);
	return HAL_I2C_Master_Receive(&hi2c1, OV7670_I2C_ADDR, val, 1, 100);
}

static void OV7670_HardReset(void) {
	/* PWDN = 0 -> tryb normalny */
	HAL_GPIO_WritePin(OV7670_PWDN_PORT, OV7670_PWDN_PIN, GPIO_PIN_RESET);
	HAL_Delay(10);

	/* RESET# = 0 -> reset, 1 -> praca */
	HAL_GPIO_WritePin(OV7670_RST_PORT, OV7670_RST_PIN, GPIO_PIN_RESET);
	HAL_Delay(20);
	HAL_GPIO_WritePin(OV7670_RST_PORT, OV7670_RST_PIN, GPIO_PIN_SET);
	HAL_Delay(100); /* datasheet: >1ms po resecie, dajemy zapas */
}

static HAL_StatusTypeDef OV7670_Probe(uint8_t *pid, uint8_t *ver, uint8_t *midh,
		uint8_t *midl) {
	if (OV7670_ReadReg(OV7670_REG_PID, pid) != HAL_OK)
		return HAL_ERROR;
	if (OV7670_ReadReg(OV7670_REG_VER, ver) != HAL_OK)
		return HAL_ERROR;
	if (OV7670_ReadReg(OV7670_REG_MIDH, midh) != HAL_OK)
		return HAL_ERROR;
	if (OV7670_ReadReg(OV7670_REG_MIDL, midl) != HAL_OK)
		return HAL_ERROR;
	return HAL_OK;
}

static HAL_StatusTypeDef OV7670_Init(void) {
	uint32_t i;

	/* reset programowy: COM7[7] = 1 */
	if (OV7670_WriteReg(0x12, 0x80) != HAL_OK) {
		return HAL_ERROR;
	}
	HAL_Delay(100);

	for (i = 0; i < OV7670_REGS_COUNT; i++) {
		if (OV7670_WriteReg(ov7670_regs[i][0], ov7670_regs[i][1]) != HAL_OK) {
			return HAL_ERROR;
		}
	}

	/* datasheet: po zmianie ustawien sensor potrzebuje ok. 10 klatek */
	HAL_Delay(300);
	return HAL_OK;
}

static HAL_StatusTypeDef OV7670_SetColorBar(uint8_t enable) {
	uint8_t xsc, ysc, com17;

	if (OV7670_ReadReg(0x70, &xsc) != HAL_OK)
		return HAL_ERROR;
	if (OV7670_ReadReg(0x71, &ysc) != HAL_OK)
		return HAL_ERROR;
	if (OV7670_ReadReg(0x42, &com17) != HAL_OK)
		return HAL_ERROR;

	if (enable) {
		/* test_pattern = (XSC[7], YSC[7]) = (1,0) -> 8-kolorowe paski */
		xsc |= 0x80;
		ysc &= ~0x80;
		com17 |= 0x08; /* dodatkowo paski z bloku DSP */
	} else {
		xsc &= ~0x80;
		ysc &= ~0x80;
		com17 &= ~0x08;
	}

	if (OV7670_WriteReg(0x70, xsc) != HAL_OK)
		return HAL_ERROR;
	if (OV7670_WriteReg(0x71, ysc) != HAL_OK)
		return HAL_ERROR;
	return OV7670_WriteReg(0x42, com17);
}

static HAL_StatusTypeDef OV7670_SetMirrorFlip(uint8_t mirror, uint8_t vflip) {
	uint8_t mvfp;

	if (OV7670_ReadReg(0x1E, &mvfp) != HAL_OK)
		return HAL_ERROR;

	mvfp &= ~0x30;
	if (mirror)
		mvfp |= 0x20; /* MVFP[5] */
	if (vflip)
		mvfp |= 0x10; /* MVFP[4] */

	return OV7670_WriteReg(0x1E, mvfp);
}

/* ==========================================================================
 * Callbacki DCMI
 * ========================================================================== */
void HAL_DCMI_FrameEventCallback(DCMI_HandleTypeDef *phdcmi) {
	(void) phdcmi;

/* ==========================================================================
 * Transmisja klatki
 *
 * Ramka na UART:
 *   [0xA5][0x5A]['O']['V'] [W_lo][W_hi] [H_lo][H_hi] [dane...]
 * Dane: RGB565, kolejnosc bajtow taka jak na magistrali (najpierw MSB).
 * ========================================================================== */
static void SendFrame(void) {
	uint8_t hdr[8];

	hdr[0] = FRAME_MAGIC_0;
	hdr[1] = FRAME_MAGIC_1;
	hdr[2] = 'O';
	hdr[3] = 'V';
	hdr[4] = (uint8_t) (OV7670_WIDTH & 0xFF);
	hdr[5] = (uint8_t) (OV7670_WIDTH >> 8);
	hdr[6] = (uint8_t) (OV7670_HEIGHT & 0xFF);
	hdr[7] = (uint8_t) (OV7670_HEIGHT >> 8);

	HAL_UART_Transmit(&huart3, hdr, sizeof(hdr), 200);
	HAL_UART_Transmit(&huart3, frame_buffer, OV7670_FRAME_BYTES, 5000);
}

/* ==========================================================================
 * Narzedzia diagnostyczne
 * ========================================================================== */
static void LOG(const char *fmt, ...) {
	char buf[160];
	va_list args;
	int len;

	va_start(args, fmt);
	len = vsnprintf(buf, sizeof(buf), fmt, args);
	va_end(args);

	if (len > 0) {
		HAL_UART_Transmit(&huart3, (uint8_t*) buf, (uint16_t) len, 200);
	}
}

static void I2C_ScanBus(void) {
	uint8_t addr;
	uint8_t found = 0;

	LOG("Skan I2C2  : ");
	for (addr = 0x02; addr < 0xFE; addr += 2) {
		if (HAL_I2C_IsDeviceReady(&hi2c1, addr, 2, 20) == HAL_OK) {
			LOG("0x%02X ", addr);
			found++;
		}
	}
	if (!found) {
		LOG("(nic nie znaleziono)");
	}
	LOG("\r\n");
}
